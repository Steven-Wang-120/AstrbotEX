cmake_minimum_required(VERSION 3.16)
project(jetmax_joint_nodes C)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_library(jetmax_position_query SHARED native/native_query.c)
target_compile_options(jetmax_position_query PRIVATE -Wall -Wextra -Werror)
install(TARGETS jetmax_position_query LIBRARY DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/joint_nodes scripts/cancel_joint scripts/joint_acceptance DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md SOURCE_PROVENANCE.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(acceptance test/test_acceptance.py TIMEOUT 120)
  ament_add_pytest_test(deadlines test/test_deadlines.py TIMEOUT 30)
  ament_add_pytest_test(unit test/test_session.py TIMEOUT 60)
  ament_add_pytest_test(transport test/test_transport.py TIMEOUT 30)
  ament_add_pytest_test(ros_interfaces test/test_ros.py TIMEOUT 90)
  ament_add_pytest_test(launch_process test/test_process.py TIMEOUT 60)
endif()
ament_package()
